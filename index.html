Module: python/adherence_analytics.py

class AdherenceAnalyticsEngine:
    def calculate_compliance_summary(self) -> Dict[str, Any]:
        total_scheduled = len(self.log_history)
        total_taken = sum(1 for log in self.log_history if log["status"] == "TAKEN")
        rate = round((total_taken / total_scheduled) * 100, 2)
        return {
            "adherence_rate_pct": rate,
            "clinical_risk_tier": "LOW_RISK" if rate >= 90 else "MODERATE_RISK"
        }

Module: cpp/schedule_optimizer.cpp

class ScheduleOptimizer {
    void runSafetyAndOptimizationAudit() {
        for (const auto& rule : interactionRules) {
            if (hasConflict(rule.drugA, rule.drugB)) {
                int gap = std::abs(medA.scheduledMinutes - medB.scheduledMinutes);
                if (gap < rule.minSeparationMinutes) {
                    reschedule(medB, rule.minSeparationMinutes - gap);
                }
            }
        }
    }
};

Module: java/MedicationScheduler.java

public class MedicationScheduler {
    public static class MedicationItem {
        public boolean isOverdue(LocalTime currentTime, int gracePeriodMinutes) {
            if (isTaken) return false;
            return currentTime.isAfter(scheduledTime.plusMinutes(gracePeriodMinutes));
        }
    }
}
