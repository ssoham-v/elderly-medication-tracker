void runSafetyAndOptimizationAudit() {
    for (const auto& rule : interactionRules) {
        if (hasConflict(rule.drugA, rule.drugB)) {
            int gap = std::abs(medA.scheduledMinutes - medB.scheduledMinutes);
            if (gap < rule.minSeparationMinutes) {
                reschedule(medB, rule.minSeparationMinutes - gap);
            }
        }
    }
}
